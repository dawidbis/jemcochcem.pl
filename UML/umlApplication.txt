package "Application.Interfaces" {
    interface IAiService {
        + AnalyzeMealPhotoAsync(byte[] photoBytes) Task<AiAnalysisResult>
        + AskCoachAsync(string prompt) Task<string>
    }

    interface IOffApiClient {
        + FetchProductByBarcode(string barcode) Task<FoodDto>
    }

    interface IRedisCache {
        + GetAsync<T>(string key) Task<T>
        + SetAsync<T>(string key, T value, TimeSpan ttl) Task
    }
}

package "Application.DTOs" {
    class DiaryDto {
        + DateTime Date
        + int TotalCalories
        + List<MealLogItemDto> Items
    }
    class MealLogItemDto {
        + string FoodName
        + decimal Grams
        + int Calories
        + MacroNutrientsDto Macros
    }
    class MacroNutrientsDto {
        + decimal Protein
        + decimal Carbs
        + decimal Fats
    }
    class FoodDto {
        + string Barcode
        + string Name
        + int CaloriesPer100g
    }
    class AiAnalysisResult {
        + int EstimatedCalories
        + string Description
    }
    class UserDto
    class MeasurementDto

    class AiMealPlanItemDto {
        + string MealType
        + string ProductName
        + int Grams
        + int Calories
        + decimal Proteins
        + decimal Carbs
        + decimal Fats
    }

    ' --- NOWE DTO WODY ---
    class WaterStatusDto #AliceBlue-LightBlue {
        + int CurrentAmountMl
        + int TargetAmountMl
        + bool IsExtraHydrationRequired
        + string AlertMessage
    }
}

package "Application.Features" {
    package "Diet" {
        class AddMealItemCommand {
            + Guid UserId
            + DateTime Date
            + Guid FoodProductId
            + decimal Grams
        }
        class AddMealItemHandler {
            + Handle(AddMealItemCommand request) Task<Unit>
        }
        class GetDailyDiaryQuery {
            + Guid UserId
            + DateTime Date
        }
        class GetDailyDiaryHandler {
            + Handle(GetDailyDiaryQuery request) Task<DiaryDto>
        }
    }

    package "Profile" {
        class CalculateTdeeCommand {
            + Guid UserId
            + decimal ActivityLevel
        }
        class CalculateTdeeHandler {
            + Handle(CalculateTdeeCommand request) Task<int>
        }
    }

    package "AI & MealPlans" {
        class GenerateMealPlanCommand {
            + Guid UserId
            + string Prompt
        }
        class GenerateMealPlanCommandHandler {
            + Handle(GenerateMealPlanCommand request) Task<Guid>
        }
        class AddAiMealToDiaryCommand {
            + Guid UserId
            + Guid MealPlanItemId
            + DateTime Date
        }
        class AddAiMealToDiaryCommandHandler {
            + Handle(AddAiMealToDiaryCommand request) Task<Guid>
        }
    }

    ' --- NOWA PACZKA TRANSAKCYJNA WODY ---
    package "Water" #AliceBlue-LightBlue {
        class LogWaterIntakeCommand {
            + Guid UserId
            + DateTime Date
            + int AmountMl
        }
        class LogWaterIntakeCommandHandler {
            - IWaterLogRepository _waterLogRepository
            + Handle(LogWaterIntakeCommand request) Task<Unit>
        }
        class GetWaterStatusQuery {
            + Guid UserId
            + DateTime Date
        }
        class GetWaterStatusQueryHandler {
            - IWaterLogRepository _waterLogRepository
            - IMealLogRepository _mealLogRepository
            - IBodyMeasurementRepository _measurementRepository
            + Handle(GetWaterStatusQuery request) Task<WaterStatusDto>
        }
    }
}

' Relacje MediatR (CQRS)
AddMealItemHandler ..> AddMealItemCommand : handles
GetDailyDiaryHandler ..> GetDailyDiaryQuery : handles
CalculateTdeeHandler ..> CalculateTdeeCommand : handles
GenerateMealPlanCommandHandler ..> GenerateMealPlanCommand : handles
AddAiMealToDiaryCommandHandler ..> AddAiMealToDiaryCommand : handles
LogWaterIntakeCommandHandler ..> LogWaterIntakeCommand : handles
GetWaterStatusQueryHandler ..> GetWaterStatusQuery : handles

' Relacje struktur DTO
DiaryDto *-- MealLogItemDto
MealLogItemDto *-- MacroNutrientsDto
GenerateMealPlanCommandHandler ..> AiMealPlanItemDto : parses
GetDailyDiaryHandler ..> DiaryDto : returns
GetWaterStatusQueryHandler ..> WaterStatusDto : returns

' Wstrzykiwanie repozytoriów domenowych przez konstruktory
Diet --> "Domain.Interfaces" : uses IMealLogRepository, IFoodRepository
Profile --> "Domain.Services" : uses TdeeCalculationService
"AI & MealPlans" --> "Domain.Interfaces" : uses IMealPlanRepository
Water --> "Domain.Interfaces" : uses IWaterLogRepository, IMealLogRepository, IBodyMeasurementRepository
